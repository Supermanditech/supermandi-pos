# SD-ONBOARD-002: SQL PROOF QUERIES
# =============================================================================

# Check global barcode mapping created
docker exec supermandi-db psql -U postgres -d supermandi -c "
  SELECT barcode, product_id, source, created_by_store_id
  FROM catalog.product_barcodes
  WHERE source = 'retailer_digitisation'
  ORDER BY created_at DESC
  LIMIT 5;
"

# Check store-scoped barcode mapping created
docker exec supermandi-db psql -U postgres -d supermandi -c "
  SELECT spb.barcode, spb.store_id, sp.display_name, sp.sell_price, sp.purchase_price, sp.digitisation_mode
  FROM catalog.store_product_barcodes spb
  JOIN catalog.store_products sp ON sp.id = spb.store_product_id
  ORDER BY spb.created_at DESC
  LIMIT 5;
"

# Check inventory ledger entry created
docker exec supermandi-db psql -U postgres -d supermandi -c "
  SELECT il.store_id, p.name, il.delta_qty, il.notes
  FROM inventory.inventory_ledger il
  JOIN catalog.products p ON p.id = il.product_id
  WHERE il.notes LIKE 'Initial stock from%'
  ORDER BY il.created_at DESC
  LIMIT 5;
"

# Check stock balances are correct
docker exec supermandi-db psql -U postgres -d supermandi -c "
  SELECT sb.store_id, p.name, sb.current_qty
  FROM inventory.stock_balances sb
  JOIN catalog.products p ON p.id = sb.product_id
  WHERE sb.store_id = 'a0000000-0000-0000-0000-000000000001'
  ORDER BY sb.updated_at DESC
  LIMIT 10;
"

# Verify seeded barcodes are in store_product_barcodes
docker exec supermandi-db psql -U postgres -d supermandi -c "
  SELECT spb.barcode, sp.display_name, sp.sell_price, sp.current_stock
  FROM catalog.store_product_barcodes spb
  JOIN catalog.store_products sp ON sp.id = spb.store_product_id
  WHERE spb.store_id = 'a0000000-0000-0000-0000-000000000001'
  AND spb.barcode LIKE '890100300%'
  ORDER BY spb.barcode
  LIMIT 10;
"
