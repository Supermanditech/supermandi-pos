# =============================================================================
# ON VM - RUN THESE COMMANDS
# =============================================================================

# 1. Check container names
docker ps --format "{{.Names}}"

# 2. Find path inside container (adjust container name as needed)
docker exec supermandi-enroll-service ls -la /app/dist/routes/v1/pos/ 2>/dev/null || \
docker exec supermandi-main ls -la /app/dist/routes/v1/pos/ 2>/dev/null || \
echo "Try different container name"

# 3. Backup existing files
docker exec supermandi-enroll-service cp /app/dist/routes/v1/pos/sales.js /app/dist/routes/v1/pos/sales.js.bak

# 4. Copy patched sales.js into container
docker cp ~/sales.js supermandi-enroll-service:/app/dist/routes/v1/pos/sales.js

# 5. For reorder-service policies.js
docker exec supermandi-reorder-service cp /app/dist/routes/policies.js /app/dist/routes/policies.js.bak
docker cp ~/policies.js supermandi-reorder-service:/app/dist/routes/policies.js

# 6. Restart containers
docker restart supermandi-enroll-service supermandi-reorder-service

# 7. Verify containers are running
docker ps

# 8. Check logs for errors
docker logs supermandi-enroll-service --tail 20
docker logs supermandi-reorder-service --tail 20
