1. Final Workflow Summary

What We Are Building

A production-grade, multi-party commerce platform with:

  1. Live Supplier Catalogue - Suppliers register, list products (inline + CSV), SuperAdmin approves
  2. SuperAdmin Verification - Mandatory approval for suppliers AND products before POS visibility
  3. SuperAdmin Margin Control - Edit SKUs, add SuperMandi margin, set BNPL eligibility
  4. POS SELL Flow - Consumer payments via UPI/Cash/DUE/Split to retailer account
  5. POS BUY Flow - Retailer purchases from supplier catalogue with UPI/BNPL/Credit payment
  6. BNPL - Buy Now Pay Later per-SKU for retailer purchases
  7. Embedded Credit - Loan offers to retailers based on POS sales data (OCEN-ready)
  8. UPI Integration - Razorpay for collect (SELL) and payout (BUY)

// CRITICAL: Visibility filter
const catalogQuery = `
  SELECT
    sp.id,
    COALESCE(sp.edited_name, p.name) AS name,
    COALESCE(sp.edited_category, p.category) AS category,
    p.primary_barcode AS barcode,
    sp.purchase_price AS supplier_price,
    sp.purchase_price + COALESCE(sp.supermandi_margin_minor, 0) AS retailer_price,
    COALESCE(sp.supermandi_margin_minor, 0) AS margin,
    s.business_name AS supplier_name,
    s.id AS supplier_id,
    sp.bnpl_eligible,
    sp.bnpl_max_days,
    sp.stock_quantity AS stock_qty,
    sp.moq
  FROM catalog.supplier_products sp
  JOIN supplier.suppliers s ON s.id = sp.supplier_id
  JOIN catalog.products p ON p.id = sp.product_id
  WHERE s.status = 'verified'              -- CRITICAL: Only verified suppliers
    AND sp.approval_status = 'approved'    -- CRITICAL: Only approved products
    AND sp.is_active = true
    AND ($1::uuid IS NULL OR sp.supplier_id IN (
      SELECT supplier_id FROM supplier.supplier_store_links
      WHERE store_id = $1 AND status = 'active'
    ))
  ORDER BY p.name
  LIMIT $2 OFFSET $3
`;

// src/services/razorpayClient.ts
import Razorpay from 'razorpay';

const razorpay = new Razorpay({
  key_id: process.env.RAZORPAY_KEY_ID!,
  key_secret: process.env.RAZORPAY_KEY_SECRET!,
});

export async function createOrder(amount: number, receipt: string) {
  return razorpay.orders.create({
    amount,
    currency: 'INR',
    receipt,
  });
}

export async function verifyPayment(orderId: string, paymentId: string) {
  return razorpay.payments.fetch(paymentId);
}

export async function createPayout(params: {
  fundAccountId: string;
  amount: number;
  mode: 'UPI' | 'IMPS' | 'NEFT';
  referenceId: string;
}) {
  return razorpay.payouts.create({
    account_number: process.env.RAZORPAY_ACCOUNT_NUMBER!,
    fund_account_id: params.fundAccountId,
    amount: params.amount,
    currency: 'INR',
    mode: params.mode,
    purpose: 'vendor_bill',
    reference_id: params.referenceId,
  });
}

Credit Scoring Query
SELECT
  s.id AS store_id,
  COALESCE(SUM(sales.total_minor), 0) AS total_gmv,
  COUNT(sales.id) AS transaction_count,
  (SELECT COUNT(*) FROM payments.bnpl_drawdowns d
   WHERE d.store_id = s.id AND d.status = 'paid') AS bnpl_paid,
  (SELECT COUNT(*) FROM payments.bnpl_drawdowns d
   WHERE d.store_id = s.id AND d.status IN ('overdue', 'defaulted')) AS bnpl_defaults
FROM platform.stores s
LEFT JOIN public.sales ON sales.store_id = s.id
  AND sales.created_at > NOW() - INTERVAL '90 days'
WHERE s.id = $1
GROUP BY s.id;
